cmake_minimum_required(VERSION 3.25)
project(atari_lab VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
find_package(PNG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ale CONFIG REQUIRED PATHS /opt/ale/lib/cmake/ale)
set(LIBTORCH_ROOT /opt/libtorch CACHE PATH "Official cxx11 ABI LibTorch distribution")
find_path(JSON_INCLUDE nlohmann/json.hpp PATHS /opt/nlohmann-json/include REQUIRED)
find_path(HTTP_INCLUDE httplib.h PATHS /opt/cpp-httplib/include REQUIRED)
# The prebuilt CUDA archive contains the runtime. No CUDA source is compiled here;
# direct native linking avoids TorchConfig's requirement for a separate nvcc SDK.
add_library(torch_native INTERFACE)
target_include_directories(torch_native SYSTEM INTERFACE "${LIBTORCH_ROOT}/include" "${LIBTORCH_ROOT}/include/torch/csrc/api/include")
target_compile_definitions(torch_native INTERFACE _GLIBCXX_USE_CXX11_ABI=1)
foreach(name torch torch_cpu torch_cuda c10 c10_cuda)
  find_library(TORCH_${name} ${name} PATHS "${LIBTORCH_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(torch_native INTERFACE "${TORCH_${name}}")
endforeach()
target_link_options(torch_native INTERFACE "LINKER:--no-as-needed")
set(CMAKE_BUILD_RPATH "${LIBTORCH_ROOT}/lib;/opt/ale/lib")
add_library(atari_io src/atomic_io.cpp)
target_include_directories(atari_io PUBLIC include)
target_link_libraries(atari_io PUBLIC Threads::Threads)
target_compile_options(atari_io PRIVATE -Wall -Wextra -Wpedantic)
add_library(atari_environment src/environment.cpp)
target_include_directories(atari_environment PUBLIC include)
target_include_directories(atari_environment SYSTEM PRIVATE /opt/ale/include)
target_link_libraries(atari_environment PUBLIC ale::ale-lib Threads::Threads)
target_compile_options(atari_environment PRIVATE -Wall -Wextra -Wpedantic)
add_library(atari_learning src/learning.cpp)
add_library(atari_games src/games.cpp)
target_include_directories(atari_games PUBLIC include "${JSON_INCLUDE}")
target_compile_options(atari_games PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(atari_learning PUBLIC include "${JSON_INCLUDE}")
target_link_libraries(atari_learning PUBLIC atari_environment atari_games atari_io torch_native OpenSSL::Crypto)
target_compile_options(atari_learning PRIVATE -Wall -Wextra -Wpedantic)
add_library(atari_shared src/shared.cpp)
target_include_directories(atari_shared PUBLIC include "${JSON_INCLUDE}")
target_link_libraries(atari_shared PUBLIC atari_learning)
target_compile_options(atari_shared PRIVATE -Wall -Wextra -Wpedantic)
add_executable(atari src/main.cpp src/server.cpp)
target_include_directories(atari PRIVATE "${HTTP_INCLUDE}")
target_link_libraries(atari PRIVATE atari_shared PNG::PNG)
if(BUILD_TESTING)
  add_executable(atari-atomic-io-test tests/atomic_io_test.cpp)
  target_link_libraries(atari-atomic-io-test PRIVATE atari_io)
  add_test(NAME atomic_io COMMAND atari-atomic-io-test)
  add_executable(atari-games-test tests/games_test.cpp)
  target_link_libraries(atari-games-test PRIVATE atari_games)
  add_test(NAME games COMMAND atari-games-test "${PROJECT_SOURCE_DIR}/games.json")
  add_executable(atari-environment-test tests/environment_test.cpp)
  target_link_libraries(atari-environment-test PRIVATE atari_environment)
  add_test(NAME environment COMMAND atari-environment-test /opt/ale/roms/pong.bin)
  add_executable(atari-learning-test tests/learning_test.cpp)
  target_link_libraries(atari-learning-test PRIVATE atari_learning)
  add_test(NAME learning COMMAND atari-learning-test)
  add_executable(atari-shared-test tests/shared_test.cpp)
  target_link_libraries(atari-shared-test PRIVATE atari_shared)
  add_test(NAME shared COMMAND atari-shared-test)
  add_executable(atari-checkpoint-check tests/checkpoint_check.cpp)
  target_link_libraries(atari-checkpoint-check PRIVATE atari_learning)
endif()
