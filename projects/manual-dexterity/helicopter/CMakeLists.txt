cmake_minimum_required(VERSION 3.25)
project(helicopter VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
find_library(HELICOPTER_LAPACK NAMES casadi-tp-openblas PATHS /opt/casadi NO_DEFAULT_PATH REQUIRED)
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS /opt/mujoco/include REQUIRED)
find_library(MUJOCO_LIBRARY mujoco HINTS /opt/mujoco/lib REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp HINTS /opt/nlohmann-json/include REQUIRED)
find_path(HTTP_INCLUDE_DIR httplib.h HINTS /opt/cpp-httplib/include REQUIRED)
# Keep validation provenance coupled to the compiled plant/controller and evaluator.
set(MODEL_SOURCE_FILES src/simulation.cpp src/rotor_model.cpp src/hinf_controller.cpp
  include/helicopter.hpp include/rotor_model.hpp include/hinf_controller.hpp include/flight_constraints.hpp
  include/mass_estimator.hpp include/mass_scaled_dynamics.hpp)
set(MODEL_FINGERPRINT "")
foreach(MODEL_SOURCE IN LISTS MODEL_SOURCE_FILES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${MODEL_SOURCE}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${MODEL_SOURCE}" SOURCE_HASH)
  string(APPEND MODEL_FINGERPRINT "${MODEL_SOURCE}:${SOURCE_HASH};")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluate.cpp")
string(SHA256 MODEL_SOURCE_SHA256 "${MODEL_FINGERPRINT}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluate.cpp" EVALUATOR_SOURCE_SHA256)
add_library(helicopter-rotor src/rotor_model.cpp)
target_include_directories(helicopter-rotor PUBLIC include ${JSON_INCLUDE_DIR})
target_compile_options(helicopter-rotor PRIVATE -Wall -Wextra -Wpedantic)
add_library(helicopter-core src/simulation.cpp src/hinf_controller.cpp)
target_include_directories(helicopter-core PUBLIC include ${JSON_INCLUDE_DIR} ${MUJOCO_INCLUDE_DIR})
target_link_libraries(helicopter-core PUBLIC ${MUJOCO_LIBRARY} helicopter-rotor ${HELICOPTER_LAPACK})
target_compile_definitions(helicopter-core PUBLIC
  HELICOPTER_MODEL_SOURCE_SHA256="${MODEL_SOURCE_SHA256}"
  HELICOPTER_EVALUATOR_SOURCE_SHA256="${EVALUATOR_SOURCE_SHA256}")
target_compile_options(helicopter-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(helicopter-server src/server.cpp)
target_include_directories(helicopter-server PRIVATE ${HTTP_INCLUDE_DIR})
target_link_libraries(helicopter-server PRIVATE helicopter-core Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(helicopter-server PRIVATE -Wall -Wextra -Wpedantic)
add_executable(helicopter-evaluate tests/evaluate.cpp)
target_link_libraries(helicopter-evaluate PRIVATE helicopter-core)
add_test(NAME flight-evaluation COMMAND helicopter-evaluate)
add_executable(helicopter-thermodynamics tests/thermodynamics.cpp)
target_link_libraries(helicopter-thermodynamics PRIVATE helicopter-rotor)
add_test(NAME thermodynamic-balances COMMAND helicopter-thermodynamics)
add_executable(helicopter-mass-estimator-test tests/mass_estimator.cpp)
target_link_libraries(helicopter-mass-estimator-test PRIVATE helicopter-rotor)
add_test(NAME causal-mass-estimator COMMAND helicopter-mass-estimator-test)
add_executable(helicopter-hinf-test tests/hinf_controller.cpp)
target_link_libraries(helicopter-hinf-test PRIVATE helicopter-core)
add_test(NAME minimum-entropy-hinf COMMAND helicopter-hinf-test)
set_tests_properties(flight-evaluation thermodynamic-balances minimum-entropy-hinf PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1;OMP_NUM_THREADS=1")
add_executable(helicopter-gpu-check src/gpu_check.cpp)
target_include_directories(helicopter-gpu-check PRIVATE ${JSON_INCLUDE_DIR})
target_link_libraries(helicopter-gpu-check PRIVATE ${CMAKE_DL_LIBS})
